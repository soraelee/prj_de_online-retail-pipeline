# 1. API 사용 범위
| 구분        | 테이블                                                            | API 목적                        |
| --------- | -------------------------------------------------------------- | ----------------------------- |
| 상태 확인     | `raw_retail_events`, `mart_daily_orders`, `mart_product_sales` | 적재 건수 확인                      |
| 일별 매출 마트  | `mart_daily_orders`                                            | 날짜별 주문/취소/매출 확인               |
| 상품별 매출 마트 | `mart_product_sales`                                           | 상품별 판매량/매출/취소율 확인             |
| 원천 이벤트    | `raw_retail_events`                                            | 특정 `invoice_no` 기준 raw 이벤트 확인 |
| 고객 차원     | `dim_customer`                                                 | 특정 고객 정보 확인                   |

# 4. `main.py`

```
import os
from datetime import date, datetime
from decimal import Decimal
from typing import Optional

import psycopg2
import psycopg2.extras
from fastapi import FastAPI, Query, HTTPException


app = FastAPI(
    title="Retail Pipeline API",
    description="Kafka/Spark/Airflow로 적재된 Retail 데이터를 조회하는 API",
    version="1.0.0",
)


DB_HOST = os.getenv("DB_HOST", "postgres")
DB_PORT = os.getenv("DB_PORT", "5432")
DB_NAME = os.getenv("DB_NAME", "retail_pipeline")
DB_USER = os.getenv("DB_USER", "postgres")
DB_PASSWORD = os.getenv("DB_PASSWORD", "postgres")


def get_connection():
    return psycopg2.connect(
        host=DB_HOST,
        port=DB_PORT,
        dbname=DB_NAME,
        user=DB_USER,
        password=DB_PASSWORD,
        cursor_factory=psycopg2.extras.RealDictCursor,
    )


def normalize_value(value):
    if isinstance(value, (datetime, date)):
        return value.isoformat()
    if isinstance(value, Decimal):
        return float(value)
    return value


def normalize_rows(rows):
    return [
        {key: normalize_value(value) for key, value in row.items()}
        for row in rows
    ]


def fetch_all(sql: str, params: dict = None):
    try:
        with get_connection() as conn:
            with conn.cursor() as cur:
                cur.execute(sql, params or {})
                rows = cur.fetchall()
                return normalize_rows(rows)
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))


def fetch_one(sql: str, params: dict = None):
    rows = fetch_all(sql, params)
    if not rows:
        return None
    return rows[0]


@app.get("/health")
def health_check():
    sql = "SELECT 1 AS ok"
    result = fetch_one(sql)

    return {
        "status": "ok",
        "db": result,
    }


@app.get("/api/v1/summary/counts")
def get_table_counts():
    """
    현재 주요 테이블에 데이터가 얼마나 적재되어 있는지 확인하는 API
    """
    sql = """
        SELECT 'raw_retail_events' AS table_name, COUNT(*) AS row_count
        FROM raw_retail_events

        UNION ALL

        SELECT 'mart_daily_orders' AS table_name, COUNT(*) AS row_count
        FROM mart_daily_orders

        UNION ALL

        SELECT 'mart_product_sales' AS table_name, COUNT(*) AS row_count
        FROM mart_product_sales
    """

    rows = fetch_all(sql)

    return {
        "data": rows
    }


@app.get("/api/v1/mart/daily-orders")
def get_daily_orders(
    start_date: Optional[str] = Query(None, example="2025-12-01"),
    end_date: Optional[str] = Query(None, example="2025-12-31"),
    limit: int = Query(30, ge=1, le=500),
    offset: int = Query(0, ge=0),
):
    """
    일별 주문/취소/매출 마트 조회
    """
    sql = """
        SELECT
            order_date,
            total_orders,
            total_cancels,
            total_quantity,
            total_revenue,
            avg_order_amount,
            cancel_rate
        FROM mart_daily_orders
        WHERE (%(start_date)s IS NULL OR order_date >= %(start_date)s::date)
          AND (%(end_date)s IS NULL OR order_date <= %(end_date)s::date)
        ORDER BY order_date DESC
        LIMIT %(limit)s OFFSET %(offset)s
    """

    rows = fetch_all(sql, {
        "start_date": start_date,
        "end_date": end_date,
        "limit": limit,
        "offset": offset,
    })

    return {
        "count": len(rows),
        "data": rows,
    }


@app.get("/api/v1/mart/product-sales")
def get_product_sales(
    stock_code: Optional[str] = Query(None, example="85123A"),
    limit: int = Query(30, ge=1, le=500),
    offset: int = Query(0, ge=0),
):
    """
    상품별 판매 마트 조회
    """
    sql = """
        SELECT
            stock_code,
            description,
            order_count,
            cancel_count,
            total_quantity,
            total_revenue,
            cancel_rate
        FROM mart_product_sales
        WHERE (%(stock_code)s IS NULL OR stock_code = %(stock_code)s)
        ORDER BY total_revenue DESC
        LIMIT %(limit)s OFFSET %(offset)s
    """

    rows = fetch_all(sql, {
        "stock_code": stock_code,
        "limit": limit,
        "offset": offset,
    })

    return {
        "count": len(rows),
        "data": rows,
    }


@app.get("/api/v1/raw/events")
def get_raw_events(
    invoice_no: Optional[str] = Query(None, example="536365"),
    customer_id: Optional[str] = Query(None, example="17850"),
    limit: int = Query(50, ge=1, le=500),
    offset: int = Query(0, ge=0),
):
    """
    raw 이벤트 조회.
    Postman에서 특정 invoice_no 또는 customer_id 기준으로 적재 데이터를 확인할 때 사용.
    """
    sql = """
        SELECT
            event_id,
            event_type,
            invoice_no,
            stock_code,
            description,
            quantity,
            unit_price,
            customer_id,
            country,
            invoice_timestamp,
            ingested_at,
            load_run_id
        FROM raw_retail_events
        WHERE (%(invoice_no)s IS NULL OR invoice_no = %(invoice_no)s)
          AND (%(customer_id)s IS NULL OR customer_id = %(customer_id)s)
        ORDER BY invoice_timestamp DESC, ingested_at DESC
        LIMIT %(limit)s OFFSET %(offset)s
    """

    rows = fetch_all(sql, {
        "invoice_no": invoice_no,
        "customer_id": customer_id,
        "limit": limit,
        "offset": offset,
    })

    return {
        "count": len(rows),
        "data": rows,
    }


@app.get("/api/v1/customers/{customer_id}")
def get_customer(customer_id: str):
    """
    고객 차원 테이블 조회
    """
    sql = """
        SELECT
            customer_id,
            country,
            first_purchase_at,
            last_purchase_at,
            total_order_count,
            total_cancel_count,
            total_revenue
        FROM dim_customer
        WHERE customer_id = %(customer_id)s
    """

    row = fetch_one(sql, {
        "customer_id": customer_id,
    })

    if row is None:
        raise HTTPException(status_code=404, detail="customer not found")

    return {
        "data": row
    }

# 5. `Dockerfile`

```
FROM python:3.11-slim

WORKDIR /app

COPY requirements.api.txt .

RUN pip install --no-cache-dir -r requirements.api.txt

COPY main.py .

CMD ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8000"]
```

  container_name: retail-api
  ports:
    - "8000:8000"

    DB_PORT: 5432
    DB_NAME: retail_pipeline
    DB_USER: postgres
    DB_PASSWORD: postgres
  depends_on:
    - postgres
  networks:
    - kafka-network
```
